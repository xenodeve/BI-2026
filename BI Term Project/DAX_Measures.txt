Actual Registrations = SUM(FactRegistrations[Registrations])

Source Records = SUM(FactRegistrations[SourceRecords])

Complete Months = DISTINCTCOUNT(FactRegistrations[Date])

Brand Count = DISTINCTCOUNT(FactRegistrations[BrandKey])

Vehicle Type Count = DISTINCTCOUNT(FactRegistrations[VehicleTypeKey])

Registrations Previous Year =
CALCULATE([Actual Registrations], DATEADD(DimDate[Date], -1, YEAR))

YoY Growth =
VAR Previous = [Registrations Previous Year]
RETURN IF(NOT ISBLANK(Previous), DIVIDE([Actual Registrations] - Previous, Previous))

Brand Share =
DIVIDE([Actual Registrations], CALCULATE([Actual Registrations], REMOVEFILTERS(DimBrand)))

Vehicle Type Share =
DIVIDE([Actual Registrations], CALCULATE([Actual Registrations], REMOVEFILTERS(DimVehicleType)))

Average Monthly Registrations = DIVIDE([Actual Registrations], [Complete Months])

Future Forecast =
IF(
    HASONEVALUE(DimSeries[SeriesKey]),
    CALCULATE(SUM(FactForecast[ForecastRegistrations]), FactForecast[DatasetSegment] = "Future"),
    BLANK()
)

Selected Model =
IF(
    HASONEVALUE(DimSeries[SeriesKey]),
    CALCULATE(SELECTEDVALUE(ModelComparison[Model]), ModelComparison[SelectedModel] = TRUE()),
    BLANK()
)

Selected Validation MAE =
IF(
    HASONEVALUE(DimSeries[SeriesKey]),
    CALCULATE(SELECTEDVALUE(ModelComparison[ValidationMAE]), ModelComparison[SelectedModel] = TRUE())
)

Selected Test MAE =
IF(
    HASONEVALUE(DimSeries[SeriesKey]),
    CALCULATE(SELECTEDVALUE(ModelComparison[TestMAE]), ModelComparison[SelectedModel] = TRUE())
)

Selected Test RMSE =
IF(
    HASONEVALUE(DimSeries[SeriesKey]),
    CALCULATE(SELECTEDVALUE(ModelComparison[TestRMSE]), ModelComparison[SelectedModel] = TRUE())
)

Selected Test R2 =
IF(
    HASONEVALUE(DimSeries[SeriesKey]),
    CALCULATE(SELECTEDVALUE(ModelComparison[TestR2]), ModelComparison[SelectedModel] = TRUE())
)

Forecast Test Actual =
IF(
    HASONEVALUE(DimSeries[SeriesKey]),
    CALCULATE(SUM(FactForecast[ActualRegistrations]), FactForecast[DatasetSegment] = "Test")
)

Forecast Test Prediction =
IF(
    HASONEVALUE(DimSeries[SeriesKey]),
    CALCULATE(SUM(FactForecast[ForecastRegistrations]), FactForecast[DatasetSegment] = "Test")
)

Test Absolute Error =
IF(HASONEVALUE(DimSeries[SeriesKey]), SUM(TestErrors[AbsoluteError]))

Raw Source Records = SUM(DataQuality[SourceRecords])

Incomplete Source Months =
COUNTROWS(FILTER(ALL(DataQuality), DataQuality[IsComplete] = FALSE()))

Missing Source Months =
VAR FirstSourceDate = MINX(ALL(DataQuality), DataQuality[Date])
VAR LastSourceDate = MAXX(ALL(DataQuality), DataQuality[Date])
VAR ExpectedMonths =
    SELECTCOLUMNS(
        FILTER(CALENDAR(FirstSourceDate, LastSourceDate), DAY([Date]) = 1),
        "MonthDate", [Date]
    )
VAR ObservedMonths = SELECTCOLUMNS(ALL(DataQuality), "MonthDate", DataQuality[Date])
RETURN COUNTROWS(EXCEPT(ExpectedMonths, ObservedMonths))
